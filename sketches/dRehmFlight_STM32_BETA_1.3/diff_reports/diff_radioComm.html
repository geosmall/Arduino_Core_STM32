100void getPPM() {  
101 unsigned long dt_ppm;  
102 int trig = digitalRead(PPM_Pin);  
103 if (trig==1) { //Only care about rising edge  
104 dt_ppm = micros() - time_ms;  
105 time_ms = micros();  
106   
107   
108 if (dt_ppm > 5000) { //Waiting for long pulse to indicate a new pulse train has arrived  
109 ppm_counter = 0;  
110 }  
111   
112 if (ppm_counter == 1) { //First pulse  
113 channel_1_raw = dt_ppm;  
114 }  
115   
116 if (ppm_counter == 2) { //Second pulse  
117 channel_2_raw = dt_ppm;  
118 }  
119   
120 if (ppm_counter == 3) { //Third pulse  
121 channel_3_raw = dt_ppm;  
122 }  
123   
124 if (ppm_counter == 4) { //Fourth pulse  
125 channel_4_raw = dt_ppm;  
126 }  
127   
128 if (ppm_counter == 5) { //Fifth pulse  
129 channel_5_raw = dt_ppm;  
130 }  
131   
132 if (ppm_counter == 6) { //Sixth pulse  
133 channel_6_raw = dt_ppm;  
134 }  
135   
136 ppm_counter = ppm_counter + 1;  
137 }
